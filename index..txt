  <script>
    const blogLink = document.querySelector('.profile-blog');
    const blogFace = blogLink.querySelector('.profile-blog-face');
    const reducedMotion = matchMedia('(prefers-reduced-motion: reduce)');

    function resetBlogMotion() {
      ['--blog-x', '--blog-y', '--blog-light-x', '--blog-light-y'].forEach(property => {
        blogLink.style.removeProperty(property);
      });
    }

    blogLink.addEventListener('pointermove', event => {
      if (event.pointerType === 'touch' || reducedMotion.matches) return;

      const rect = blogLink.getBoundingClientRect();
      const x = Math.max(0, Math.min(1, (event.clientX - rect.left) / rect.width));
      const y = Math.max(0, Math.min(1, (event.clientY - rect.top) / rect.height));
      blogLink.style.setProperty('--blog-x', ((x - 0.5) * 12) + 'px');
      blogLink.style.setProperty('--blog-y', ((y - 0.5) * 12) + 'px');
      blogLink.style.setProperty('--blog-light-x', (x * 100) + '%');
      blogLink.style.setProperty('--blog-light-y', (y * 100) + '%');
    });

    blogLink.addEventListener('pointerleave', resetBlogMotion);
    blogLink.addEventListener('pointercancel', resetBlogMotion);
    reducedMotion.addEventListener('change', () => {
      resetBlogMotion();
      blogFace.querySelectorAll('.profile-blog-ripple').forEach(ripple => ripple.remove());
    });

    blogLink.addEventListener('pointerdown', event => {
      if (event.button !== 0 || reducedMotion.matches) return;

      const rect = blogFace.getBoundingClientRect();
      const ripple = document.createElement('span');
      ripple.className = 'profile-blog-ripple';
      ripple.setAttribute('aria-hidden', 'true');
      ripple.style.left = (event.clientX - rect.left) + 'px';
      ripple.style.top = (event.clientY - rect.top) + 'px';
      ripple.addEventListener('animationend', () => ripple.remove(), { once: true });
      blogFace.append(ripple);
    });
  </script>
